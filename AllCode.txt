	void HeapSwap(TArray<int32>& InHeap, TArray<int32>& InPositions, int32 InA, int32 InB)
	{
		const int32 NodeA = InHeap[InA];
		const int32 NodeB = InHeap[InB];
		InHeap[InA] = NodeB;
		InHeap[InB] = NodeA;
		InPositions[NodeA] = InB;
		InPositions[NodeB] = InA;
	}

	void HeapSiftUp(TArray<int32>& InHeap, TArray<int32>& InPositions, int32 InIndex, const TArray<float>& InFScores)
	{
		while (InIndex > 0)
		{
			const int32 Parent = (InIndex - 1) / 2;
			if (InFScores[InHeap[InIndex]] >= InFScores[InHeap[Parent]]) break;
			HeapSwap(InHeap, InPositions, InIndex, Parent);
			InIndex = Parent;
		}
	}

	void HeapSiftDown(TArray<int32>& InHeap, TArray<int32>& InPositions, int32 InIndex, const TArray<float>& InFScores)
	{
		const int32 Count = InHeap.Num();
		while (true)
		{
			const int32 Left = InIndex * 2 + 1;
			const int32 Right = Left + 1;
			int32 Smallest = InIndex;

			if (Left < Count && InFScores[InHeap[Left]] < InFScores[InHeap[Smallest]]) Smallest = Left;
			if (Right < Count && InFScores[InHeap[Right]] < InFScores[InHeap[Smallest]]) Smallest = Right;
			if (Smallest == InIndex) break;

			HeapSwap(InHeap, InPositions, InIndex, Smallest);
			InIndex = Smallest;
		}
	}

	void HeapPush(TArray<int32>& InHeap, TArray<int32>& InPositions, int32 InNode, const TArray<float>& InFScores)
	{
		InPositions[InNode] = InHeap.Num();
		InHeap.Add(InNode);
		HeapSiftUp(InHeap, InPositions, InHeap.Num() - 1, InFScores);
	}

	int32 HeapPop(TArray<int32>& InHeap, TArray<int32>& InPositions, const TArray<float>& InFScores)
	{
		const int32 Top = InHeap[0];
		InPositions[Top] = INDEX_NONE;

		const int32 LastIndex = InHeap.Num() - 1;
		if (LastIndex > 0)
		{
			InHeap[0] = InHeap[LastIndex];
			InPositions[InHeap[0]] = 0;
		}
		InHeap.RemoveAt(LastIndex);

		if (InHeap.Num() > 0)
		{
			HeapSiftDown(InHeap, InPositions, 0, InFScores);
		}
		return Top;
	}

	void HeapUpdate(TArray<int32>& InHeap, TArray<int32>& InPositions, int32 InNode, const TArray<float>& InFScores)
	{
		const int32 Index = InPositions[InNode];
		if (Index == INDEX_NONE) return;
		HeapSiftUp(InHeap, InPositions, Index, InFScores);
		HeapSiftDown(InHeap, InPositions, InPositions[InNode], InFScores);
	}
}

FRailPathResult UAStarPathFinder::FindPath(
	const FRailGridData& InGrid,
	int32 InStartIndex,
	int32 InEndIndex,
	int32 InMaxExpansions)
{
	FRailPathResult Result;

	if (!CostCalculator)
	{
		CostCalculator = NewObject<UCostCalculator>(this);
	}

	const int32 NumNodes = InGrid.Nodes.Num();
	if (NumNodes == 0 || !InGrid.Nodes.IsValidIndex(InStartIndex) || !InGrid.Nodes.IsValidIndex(InEndIndex))
	{
		UE_LOG(LogTemp, Error, TEXT("[AStar] Некорректные индексы: %d -> %d (всего %d)."),
			InStartIndex, InEndIndex, NumNodes);
		return Result;
	}

	const FRailGridNode& StartNode = InGrid.Nodes[InStartIndex];
	const FRailGridNode& EndNode = InGrid.Nodes[InEndIndex];

	if (StartNode.bIsForbidden || EndNode.bIsForbidden)
	{
		UE_LOG(LogTemp, Warning, TEXT("[AStar] Старт или финиш в запретной зоне."));
		return Result;
	}

	if (InStartIndex == InEndIndex)
	{
		Result.bFound = true;
		Result.NodeIndices.Add(InStartIndex);
		return Result;
	}

	TArray<float> GScore;
	TArray<float> FScore;
	TArray<int32> CameFrom;
	TArray<int8>  NodeState;
	TArray<int32> OpenHeap;
	TArray<int32> HeapPositions;

	GScore.Init(TNumericLimits<float>::Max(), NumNodes);
	FScore.Init(TNumericLimits<float>::Max(), NumNodes);
	CameFrom.Init(INDEX_NONE, NumNodes);
	NodeState.Init(0, NumNodes);
	HeapPositions.Init(INDEX_NONE, NumNodes);

	GScore[InStartIndex] = 0.0f;
	FScore[InStartIndex] = Heuristic(InGrid, InStartIndex, InEndIndex);
	HeapPush(OpenHeap, HeapPositions, InStartIndex, FScore);
	NodeState[InStartIndex] = 1;

	TArray<int32, TInlineAllocator<8>> Neighbors;
	int32 Expansions = 0;

	while (OpenHeap.Num() > 0)
	{
		if (++Expansions > InMaxExpansions)
		{
			UE_LOG(LogTemp, Warning, TEXT("[AStar] Превышен лимит расширений (%d)."), InMaxExpansions);
			break;
		}

		const int32 CurrentIndex = HeapPop(OpenHeap, HeapPositions, FScore);

		if (CurrentIndex == InEndIndex)
		{
			Result.bFound = true;
			Result.ExpandedNodes = Expansions;
			ReconstructPath(InGrid, CameFrom, InStartIndex, InEndIndex, Result);
			return Result;
		}

		NodeState[CurrentIndex] = 2;

		Neighbors.Reset();
		CollectNeighbors(InGrid, CurrentIndex, Neighbors);

		for (const int32 NeighborIndex : Neighbors)
		{
			if (NodeState[NeighborIndex] == 2) continue;

			const float EdgeCost = EvaluateEdge(InGrid, CurrentIndex, NeighborIndex, CameFrom);
			const float TentativeG = GScore[CurrentIndex] + EdgeCost;

			if (TentativeG < GScore[NeighborIndex])
			{
				CameFrom[NeighborIndex] = CurrentIndex;
				GScore[NeighborIndex] = TentativeG;
				FScore[NeighborIndex] = TentativeG + Heuristic(InGrid, NeighborIndex, InEndIndex);

				if (NodeState[NeighborIndex] == 1)
				{
					HeapUpdate(OpenHeap, HeapPositions, NeighborIndex, FScore);
				}
				else
				{
					HeapPush(OpenHeap, HeapPositions, NeighborIndex, FScore);
					NodeState[NeighborIndex] = 1;
				}
			}
		}
	}

	Result.ExpandedNodes = Expansions;
	UE_LOG(LogTemp, Warning, TEXT("[AStar] Путь не найден. Расширений: %d."), Expansions);
	return Result;
}

{
	CostCalculator = InCalculator;
}

void UAStarPathFinder::CollectNeighbors(
	const FRailGridData& InGrid,
	int32 InNodeIndex,
	TArray<int32, TInlineAllocator<8>>& OutNeighbors) const
{
	const FRailGridNode& Node = InGrid.Nodes[InNodeIndex];

	for (int32 i = 0; i < NumDirections; ++i)
	{
		const int32 NX = Node.X + DirectionX[i];
		const int32 NY = Node.Y + DirectionY[i];

		if (!InGrid.IsValid(NX, NY)) continue;

		const int32 NeighborIndex = InGrid.IndexOf(NX, NY);
		if (InGrid.Nodes[NeighborIndex].bIsForbidden) continue;

		if (DirectionX[i] != 0 && DirectionY[i] != 0)
		{
			const int32 SideA = InGrid.IndexOf(Node.X + DirectionX[i], Node.Y);
			const int32 SideB = InGrid.IndexOf(Node.X, Node.Y + DirectionY[i]);

			if (InGrid.Nodes[SideA].bIsForbidden || InGrid.Nodes[SideB].bIsForbidden) continue;
		}

		OutNeighbors.Add(NeighborIndex);
	}
}

void UAStarPathFinder::ReconstructPath(
	const FRailGridData& InGrid,
	const TArray<int32>& InCameFrom,
	int32 InStartIndex,
	int32 InEndIndex,
	FRailPathResult& OutResult) const
{
	int32 Current = InEndIndex;
	while (Current != INDEX_NONE)
	{
		OutResult.NodeIndices.Add(Current);
		if (Current == InStartIndex) break;
		Current = InCameFrom[Current];
	}
	Algo::Reverse(OutResult.NodeIndices);

	const int32 NumEdges = FMath::Max(0, OutResult.NodeIndices.Num() - 1);
	OutResult.EdgeCosts.Reserve(NumEdges);
	OutResult.EdgeLengths.Reserve(NumEdges);

	for (int32 i = 1; i < OutResult.NodeIndices.Num(); ++i)
	{
		const int32 PrevIndex = OutResult.NodeIndices[i - 1];
		const int32 CurrIndex = OutResult.NodeIndices[i];

		const FRailGridNode& PrevNode = InGrid.Nodes[PrevIndex];
		const FRailGridNode& CurrNode = InGrid.Nodes[CurrIndex];

		const int32 DX = FMath::Abs(CurrNode.X - PrevNode.X);
		const int32 DY = FMath::Abs(CurrNode.Y - PrevNode.Y);
		const bool bDiagonal = (DX != 0 && DY != 0);

		const float EdgeLength_m = bDiagonal
			? InGrid.StepSize_m * FMath::Sqrt(2.0f)
			: InGrid.StepSize_m;

		const float SlopeRatio = FMath::Abs(CurrNode.Height_m - PrevNode.Height_m) / EdgeLength_m;

		const float TurnAngleDeg = (i >= 2)
			? [&]() -> float
			{
				const FRailGridNode& BeforeNode = InGrid.Nodes[OutResult.NodeIndices[i - 2]];
				const float PrevDirX = PrevNode.X - BeforeNode.X;
				const float PrevDirY = PrevNode.Y - BeforeNode.Y;
				const float CurrDirX = CurrNode.X - PrevNode.X;
				const float CurrDirY = CurrNode.Y - PrevNode.Y;

				const float Dot = PrevDirX * CurrDirX + PrevDirY * CurrDirY;
				const float LenA = FMath::Sqrt(PrevDirX * PrevDirX + PrevDirY * PrevDirY);
				const float LenB = FMath::Sqrt(CurrDirX * CurrDirX + CurrDirY * CurrDirY);
				if (LenA < KINDA_SMALL_NUMBER || LenB < KINDA_SMALL_NUMBER) return 0.0f;

				const float CosAngle = FMath::Clamp(Dot / (LenA * LenB), -1.0f, 1.0f);
				return FMath::RadiansToDegrees(FMath::Acos(CosAngle));
			}()
				: 0.0f;

			const float GroundCoef = CostCalculator->GetGroundCoefficient(CurrNode.GroundType);

			// Уклон — в процентах, как требует формула ТЗ.
			const float SlopePercent = SlopeRatio * 100.0f;

			const float EdgeCost = CostCalculator->CalculateEdgeCost(
				EdgeLength_m, SlopePercent, TurnAngleDeg, GroundCoef);

			OutResult.EdgeCosts.Add(EdgeCost);
			OutResult.EdgeLengths.Add(EdgeLength_m);
			OutResult.TotalLength_m += EdgeLength_m;
			OutResult.TotalCost += EdgeCost;
	}

	// Диагностика — короткая сводка.
	UE_LOG(LogTemp, Verbose,
		TEXT("[AStar] Reconstruct: узлов=%d, рёбер=%d, длина=%.1f м, стоимость=%.1f ед."),
		OutResult.NodeIndices.Num(), OutResult.EdgeCosts.Num(),
		OutResult.TotalLength_m, OutResult.TotalCost);
}

float UAStarPathFinder::EvaluateEdge(
	const FRailGridData& InGrid,
	int32 InFromIndex,
	int32 InToIndex,
	const TArray<int32>& InCameFrom) const
{
	const FRailGridNode& FromNode = InGrid.Nodes[InFromIndex];
	const FRailGridNode& ToNode = InGrid.Nodes[InToIndex];

	const int32 DX = FMath::Abs(ToNode.X - FromNode.X);
	const int32 DY = FMath::Abs(ToNode.Y - FromNode.Y);
	const bool bDiagonal = (DX != 0 && DY != 0);

	const float EdgeLength_m = bDiagonal
		? InGrid.StepSize_m * FMath::Sqrt(2.0f)
		: InGrid.StepSize_m;

	const float SlopeRatio = FMath::Abs(ToNode.Height_m - FromNode.Height_m) / EdgeLength_m;

	float TurnAngleDeg = 0.0f;
	const int32 PrevIndex = InCameFrom[InFromIndex];
	if (PrevIndex != INDEX_NONE)
	{
		const FRailGridNode& PrevNode = InGrid.Nodes[PrevIndex];
		const float PrevDirX = FromNode.X - PrevNode.X;
		const float PrevDirY = FromNode.Y - PrevNode.Y;
		const float CurrDirX = ToNode.X - FromNode.X;
		const float CurrDirY = ToNode.Y - FromNode.Y;

		const float Dot = PrevDirX * CurrDirX + PrevDirY * CurrDirY;
		const float LenA = FMath::Sqrt(PrevDirX * PrevDirX + PrevDirY * PrevDirY);
		const float LenB = FMath::Sqrt(CurrDirX * CurrDirX + CurrDirY * CurrDirY);

		if (LenA > KINDA_SMALL_NUMBER && LenB > KINDA_SMALL_NUMBER)
		{
			const float CosAngle = FMath::Clamp(Dot / (LenA * LenB), -1.0f, 1.0f);
			TurnAngleDeg = FMath::RadiansToDegrees(FMath::Acos(CosAngle));
		}
	}

	const float GroundCoef = CostCalculator->GetGroundCoefficient(ToNode.GroundType);
	const float SlopePercent = SlopeRatio * 100.0f;

	return CostCalculator->CalculateEdgeCost(
		EdgeLength_m, SlopePercent, TurnAngleDeg, GroundCoef);
}

float UAStarPathFinder::Heuristic(
	const FRailGridData& InGrid,
	int32 InFromIndex,
	int32 InToIndex) const
{
	const FRailGridNode& From = InGrid.Nodes[InFromIndex];
	const FRailGridNode& To = InGrid.Nodes[InToIndex];

	const float DX = static_cast<float>(To.X - From.X) * InGrid.StepSize_m;
	const float DY = static_cast<float>(To.Y - From.Y) * InGrid.StepSize_m;
	const float Distance = FMath::Sqrt(DX * DX + DY * DY);

	const float MinCoefficient = CostCalculator ? CostCalculator->GetMinGroundCoefficient() : 1.0f;
	return Distance * MinCoefficient;
}

UCostCalculator::UCostCalculator()
{
	GroundCoefficients = GetDefaultCoefficients();
	UpdateMinCoefficient();
}

// ─────────────────────────────────────────────────────────────────
//  Геттеры
// ─────────────────────────────────────────────────────────────────

float UCostCalculator::CalculateEdgeCost(
	float InLength_m,
	float InSlopeRatio,
	float InTurnAngleDeg,
	float InGroundCoefficient) const
{
	const float SlopeFactor = 1.0f + FMath::Abs(InSlopeRatio) * Parameters.SlopePenalty;
	const float TurnFactor = 1.0f + Parameters.TurnPenalty * FMath::Square(InTurnAngleDeg / 90.0f);

	return InLength_m * SlopeFactor * InGroundCoefficient * TurnFactor;
}

float UCostCalculator::GetGroundCoefficient(ERailGroundType InGroundType) const
{
	const int32 Index = static_cast<int32>(InGroundType);
	if (!GroundCoefficients.IsValidIndex(Index))
	{
		return 1.0f;
	}
	return GroundCoefficients[Index];
}

bool UCostCalculator::IsForbiddenGround(ERailGroundType InGroundType) const
{
	return InGroundType == ERailGroundType::City
		|| InGroundType == ERailGroundType::Reserve;
}

// ─────────────────────────────────────────────────────────────────
//  Управление состоянием
// ─────────────────────────────────────────────────────────────────

void UCostCalculator::SetGroundCoefficient(ERailGroundType InGroundType, float InCoefficient)
{
	const int32 Index = static_cast<int32>(InGroundType);
	if (!GroundCoefficients.IsValidIndex(Index))
	{
		return;
	}

	GroundCoefficients[Index] = FMath::Max(InCoefficient, 0.01f);
	UpdateMinCoefficient();
}

// ─────────────────────────────────────────────────────────────────
//  Внутренняя логика
// ─────────────────────────────────────────────────────────────────

void UCostCalculator::UpdateMinCoefficient()
{
	MinGroundCoefficient = TNumericLimits<float>::Max();
	for (const float Coefficient : GroundCoefficients)
	{
		MinGroundCoefficient = FMath::Min(MinGroundCoefficient, Coefficient);
	}
}

//  Константы
// ─────────────────────────────────────────────────────────────────

const TArray<float>& UCostCalculator::GetDefaultCoefficients()
{
	// Значения откалиброваны так, чтобы:
	//  - песок и гравий дешевле суглинка;
	//  - скала и болото — самые дорогие;
	//  - город и заповедник — фактически запретные зоны.
	static const TArray<float> Defaults = {
		1.0f,   // Loam
		0.9f,   // Sand
		3.5f,   // Rock
		5.0f,   // Water
		0.8f,   // Gravel
		1.2f,   // Clay
		1.8f,   // Forest
		10.0f,  // City
		10.0f   // Reserve
	};
	return Defaults;
}

void UHeightInterpolator::Initialize(const FRailHeightMapData& InMapData)
{
	MapData = InMapData;
	ElevationRange_m = MapData.MaxElevation_m - MapData.MinElevation_m;

	const int32 ExpectedSize = MapData.Width * MapData.Height;
	if (MapData.Width <= 0 || MapData.Height <= 0 || MapData.Heights.Num() != ExpectedSize)
	{
		UE_LOG(LogTemp, Error, TEXT("[HeightInterpolator] Некорректные данные карты: %dx%d, пикселей %d."),
			MapData.Width, MapData.Height, MapData.Heights.Num());
		bInitialized = false;
		return;
	}

	bInitialized = true;
	UE_LOG(LogTemp, Log, TEXT("[HeightInterpolator] Инициализация %dx%d, размах %.1f м."),
		MapData.Width, MapData.Height, ElevationRange_m);
}

void UHeightInterpolator::SetWorldMapping(const FVector2D& InWorldOrigin, const FVector2D& InWorldSizeXY, float InWorldSizeZ)
{
	WorldOrigin = InWorldOrigin;
	WorldSizeXY = (InWorldSizeXY.X > KINDA_SMALL_NUMBER && InWorldSizeXY.Y > KINDA_SMALL_NUMBER)
		? InWorldSizeXY
		: FVector2D(1.0f, 1.0f);
	WorldSizeZ = FMath::Max(InWorldSizeZ, KINDA_SMALL_NUMBER);
}

// ─────────────────────────────────────────────────────────────────
//  Геттеры
// ─────────────────────────────────────────────────────────────────

float UHeightInterpolator::GetHeightNormalized(float InNormX, float InNormY) const
{
	if (!bInitialized)
	{
		return 0.0f;
	}

	const float PixelX = FMath::Clamp(InNormX, 0.0f, 1.0f) * (MapData.Width - 1);
	const float PixelY = FMath::Clamp(InNormY, 0.0f, 1.0f) * (MapData.Height - 1);

	return SampleBilinear(PixelX, PixelY);
}

float UHeightInterpolator::GetHeightPixel(float InPixelX, float InPixelY) const
{
	if (!bInitialized)
	{
		return 0.0f;
	}

	const float PixelX = FMath::Clamp(InPixelX, 0.0f, static_cast<float>(MapData.Width - 1));
	const float PixelY = FMath::Clamp(InPixelY, 0.0f, static_cast<float>(MapData.Height - 1));

	return SampleBilinear(PixelX, PixelY);
}

float UHeightInterpolator::GetHeightWorld(const FVector& InWorldLocation) const
{
	if (!bInitialized)
	{
		return 0.0f;
	}

	const FVector2D Pixel = WorldToPixel(InWorldLocation);
	return SampleBilinear(Pixel.X, Pixel.Y);
}

float UHeightInterpolator::GetSlopeRatio(const FVector& InFrom, const FVector& InTo) const
{
	const float HeightFrom = GetHeightWorld(InFrom);
	const float HeightTo = GetHeightWorld(InTo);

	const FVector2D DeltaXY(InTo.X - InFrom.X, InTo.Y - InFrom.Y);
	const float DistanceXY = DeltaXY.Size();

	if (DistanceXY < KINDA_SMALL_NUMBER)
	{
		return 0.0f;
	}

	// Расстояние UE-единицы (см) → метры.
	const float DistanceXY_m = DistanceXY * 0.01f;
	return (HeightTo - HeightFrom) / DistanceXY_m;
}

float UHeightInterpolator::GetSlopeDegrees(const FVector& InFrom, const FVector& InTo) const
{
	const float Ratio = GetSlopeRatio(InFrom, InTo);
	return FMath::RadiansToDegrees(FMath::Atan(Ratio));
}

// ─────────────────────────────────────────────────────────────────
//  Внутренняя логика
// ─────────────────────────────────────────────────────────────────

float UHeightInterpolator::SampleBilinear(float InPixelX, float InPixelY) const
{
	const int32 X0 = FMath::FloorToInt(InPixelX);
	const int32 Y0 = FMath::FloorToInt(InPixelY);
	const int32 X1 = X0 + 1;
	const int32 Y1 = Y0 + 1;

	const float FracX = InPixelX - static_cast<float>(X0);
	const float FracY = InPixelY - static_cast<float>(Y0);

	const float H00 = GetPixelClamped(X0, Y0);
	const float H10 = GetPixelClamped(X1, Y0);
	const float H01 = GetPixelClamped(X0, Y1);
	const float H11 = GetPixelClamped(X1, Y1);

	const float H0 = FMath::Lerp(H00, H10, FracX);
	const float H1 = FMath::Lerp(H01, H11, FracX);
	const float NormHeight = FMath::Lerp(H0, H1, FracY);

	return MapData.MinElevation_m + NormHeight * ElevationRange_m;
}

float UHeightInterpolator::GetPixelClamped(int32 InX, int32 InY) const
{
	const int32 ClampedX = FMath::Clamp(InX, 0, MapData.Width - 1);
	const int32 ClampedY = FMath::Clamp(InY, 0, MapData.Height - 1);
	return MapData.Heights[ClampedY * MapData.Width + ClampedX];
}

FVector2D UHeightInterpolator::WorldToPixel(const FVector& InWorldLocation) const
{
	const float NormX = (InWorldLocation.X - WorldOrigin.X) / WorldSizeXY.X;

// ─────────────────────────────────────────────────────────────────
//  Внутренняя логика
// ─────────────────────────────────────────────────────────────────

bool UMapLoader::DecodeImage(const TArray<uint8>& InRawBytes, FRailHeightMapData& OutData)
{
	IImageWrapperModule& ImageWrapperModule =
		FModuleManager::LoadModuleChecked<IImageWrapperModule>(FName("ImageWrapper"));

	TSharedPtr<IImageWrapper> Wrapper =
		ImageWrapperModule.CreateImageWrapper(EImageFormat::PNG);

	if (!Wrapper.IsValid() || !Wrapper->SetCompressed(InRawBytes.GetData(), InRawBytes.Num()))
	{
		UE_LOG(LogTemp, Error, TEXT("[MapLoader] PNG не распознан."));
		return false;
	}

	OutData.Width = Wrapper->GetWidth();
	OutData.Height = Wrapper->GetHeight();

	TArray64<uint8> RawData;
	if (!Wrapper->GetRaw(ERGBFormat::Gray, 16, RawData))
	{
		UE_LOG(LogTemp, Error, TEXT("[MapLoader] Не удалось получить 16-бит raw-данные."));
		return false;
	}

	const int32 NumPixels = OutData.Width * OutData.Height;
	OutData.Heights.SetNumUninitialized(NumPixels);

	const uint16* Ptr16 = reinterpret_cast<const uint16*>(RawData.GetData());
	for (int32 i = 0; i < NumPixels; ++i)
	{
		OutData.Heights[i] = static_cast<float>(Ptr16[i]) / 65535.0f;
	}

	UE_LOG(LogTemp, Log, TEXT("[MapLoader] Загружено %dx%d (16-bit)."), OutData.Width, OutData.Height);
	return true;
}

{

	FRailHeightMapData HeightMap;
	if (!Loader->LoadHeightMap(InHeightMapPath, HeightMap))

	}

	HeightMap.MinElevation_m = TerrainParameters.MinElevation_m;

		TEXT("[ProceduralTerrain] OriginLat=%.6f, OriginLon=%.6f, PixelSize=%.2f"),
		HeightMap.OriginLat, HeightMap.OriginLon, HeightMap.PixelSize_m);

			static_cast<float>(i / Resolution) / static_cast<float>(Resolution - 1));

// ─────────────────────────────────────────────────────────────────
//  SampleTerrainZ_cm — билинейная интерполяция по сетке меша
// ─────────────────────────────────────────────────────────────────

bool AProceduralTerrain::SampleTerrainZ_cm(const FVector2D& InWorldXY, float& OutZ_cm) const
{
	const int32 Res = TerrainParameters.MeshResolution;
	const float Step_cm = TerrainParameters.StepSize_m * 100.0f;

	if (Res < 2 || Step_cm <= 0.0f) return false;
	if (MeshVertexHeightsZ_cm.Num() != Res * Res) return false;

	// Мировые XY → дробные индексы сетки.
	const float FX = InWorldXY.X / Step_cm;
	const float FY = InWorldXY.Y / Step_cm;

	// Допускаем попадание точно на границу (после clamp'а).
	const float MaxIdx = static_cast<float>(Res - 1);
	if (FX < 0.0f || FY < 0.0f || FX > MaxIdx + 0.5f || FY > MaxIdx + 0.5f) return false;

	const int32 X0 = FMath::Clamp(FMath::FloorToInt(FX), 0, Res - 1);
	const int32 Y0 = FMath::Clamp(FMath::FloorToInt(FY), 0, Res - 1);
	const int32 X1 = FMath::Min(X0 + 1, Res - 1);
	const int32 Y1 = FMath::Min(Y0 + 1, Res - 1);

	const float FracX = FX - X0;
	const float FracY = FY - Y0;

	const float H00 = MeshVertexHeightsZ_cm[Y0 * Res + X0];
	const float H10 = MeshVertexHeightsZ_cm[Y0 * Res + X1];
	const float H01 = MeshVertexHeightsZ_cm[Y1 * Res + X0];
	const float H11 = MeshVertexHeightsZ_cm[Y1 * Res + X1];

	const float H0 = FMath::Lerp(H00, H10, FracX);
	const float H1 = FMath::Lerp(H01, H11, FracX);

	OutZ_cm = FMath::Lerp(H0, H1, FracY);
	return true;
}

bool AProceduralTerrain::LoadRegionMetadataFromFile(const FString& InFullPath)
{
	FString RawJson;
	if (!FFileHelper::LoadFileToString(RawJson, *InFullPath))
	{
		UE_LOG(LogTemp, Warning,
			TEXT("[ProceduralTerrain] Не удалось прочитать: %s"), *InFullPath);
		return false;
	}

	TSharedPtr<FJsonObject> Root;
	const TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(RawJson);

	if (!FJsonSerializer::Deserialize(Reader, Root) || !Root.IsValid())
	{
		UE_LOG(LogTemp, Warning,
			TEXT("[ProceduralTerrain] Ошибка парсинга JSON: %s"), *InFullPath);
		return false;
	}

	double MinElev = TerrainParameters.MinElevation_m;
	double MaxElev = TerrainParameters.MaxElevation_m;

	Root->TryGetNumberField(TEXT("min_elevation_m"), MinElev);
	Root->TryGetNumberField(TEXT("max_elevation_m"), MaxElev);

	TerrainParameters.MinElevation_m = static_cast<float>(MinElev);
	TerrainParameters.MaxElevation_m = static_cast<float>(MaxElev);

	double WidthPxD = 0.0;
	double HeightPxD = 0.0;
	double PixelSizeD = 30.0;

	Root->TryGetNumberField(TEXT("width_px"), WidthPxD);
	Root->TryGetNumberField(TEXT("height_px"), HeightPxD);
	Root->TryGetNumberField(TEXT("pixel_size_m"), PixelSizeD);

	SourceWidthPx = static_cast<int32>(WidthPxD);
	SourcePixelSize_m = static_cast<float>(PixelSizeD);

	if (SourceWidthPx <= 0)
	{
		UE_LOG(LogTemp, Warning,
			TEXT("[ProceduralTerrain] width_px отсутствует. MeshResolution=%d."),
			TerrainParameters.MeshResolution);
		return true;
	}

	const int32 MaxRes = FMath::Max(TerrainParameters.MaxMeshResolution, 64);

	int32 TargetRes = SourceWidthPx;

	if (TargetRes > MaxRes)
	{
		TargetRes = MaxRes;

		for (int32 Div = 2; Div <= 32; ++Div)
		{
			if (SourceWidthPx % Div == 0)
			{
				const int32 Candidate = SourceWidthPx / Div;
				if (Candidate <= MaxRes)
				{
					TargetRes = Candidate;
					break;
				}
			}
		}
	}

	TerrainParameters.MeshResolution = TargetRes;

	const float SourceMapSizeMeters = SourceWidthPx * SourcePixelSize_m;
	TerrainParameters.StepSize_m = SourceMapSizeMeters / TargetRes;

	UE_LOG(LogTemp, Log,
		TEXT("[ProceduralTerrain] MeshResolution=%d (source=%d, max=%d), ")
		TEXT("StepSize=%.2f м, размер карты=%.1f км"),
		TargetRes, SourceWidthPx, MaxRes,
		TerrainParameters.StepSize_m,
		SourceMapSizeMeters / 1000.0f);

	const TSharedPtr<FJsonObject>* BboxObj = nullptr;
	if (Root->TryGetObjectField(TEXT("bbox"), BboxObj) && BboxObj)
	{
		double West = 0.0, South = 0.0;
		(*BboxObj)->TryGetNumberField(TEXT("west"), West);
		(*BboxObj)->TryGetNumberField(TEXT("south"), South);

		PendingOriginLat = South;
		PendingOriginLon = West;

		UE_LOG(LogTemp, Log,
			TEXT("[ProceduralTerrain] bbox: OriginLat=%.6f, OriginLon=%.6f"),
			PendingOriginLat, PendingOriginLon);
	}
	else
	{
		UE_LOG(LogTemp, Warning,
			TEXT("[ProceduralTerrain] В region_meta.json нет 'bbox'."));
	}

	UE_LOG(LogTemp, Log,
		TEXT("[ProceduralTerrain] Загружено из region_meta.json: ")
		TEXT("MinElev=%.1f, MaxElev=%.1f, WidthPx=%d, PixelSize=%.2f"),
		TerrainParameters.MinElevation_m,
		TerrainParameters.MaxElevation_m,
		SourceWidthPx, SourcePixelSize_m);

	return true;
}

void AProceduralTerrain::BuildMeshData(
	const FRailHeightMapData& InHeightMap,
	int32 InResolution,
	TArray<FVector>& OutVertices,
	TArray<int32>& OutTriangles,
	TArray<FLinearColor>& OutColors,
	TArray<FVector>& OutNormals) const
{
	const int32 NumVertices = InResolution * InResolution;
	const float StepSize = TerrainParameters.StepSize_m;
	const float ZScale = TerrainParameters.ZScale;
	const float ElevRange = FMath::Max(
		InHeightMap.MaxElevation_m - InHeightMap.MinElevation_m, 1.0f);

	OutVertices.Reserve(NumVertices);
	OutColors.Reserve(NumVertices);
	OutNormals.Reserve(NumVertices);
	OutTriangles.Reserve((InResolution - 1) * (InResolution - 1) * 6);

	const float ScaleX = static_cast<float>(InHeightMap.Width - 1)
		/ FMath::Max(InResolution - 1, 1);
	const float ScaleY = static_cast<float>(InHeightMap.Height - 1)
		/ FMath::Max(InResolution - 1, 1);

	for (int32 Y = 0; Y < InResolution; ++Y)
	{
		for (int32 X = 0; X < InResolution; ++X)
		{
			const float SrcX = X * ScaleX;
			const float SrcY = Y * ScaleY;

			const int32 X0 = FMath::Clamp(FMath::FloorToInt(SrcX), 0, InHeightMap.Width - 1);
			const int32 Y0 = FMath::Clamp(FMath::FloorToInt(SrcY), 0, InHeightMap.Height - 1);
			const int32 X1 = FMath::Clamp(X0 + 1, 0, InHeightMap.Width - 1);
			const int32 Y1 = FMath::Clamp(Y0 + 1, 0, InHeightMap.Height - 1);

			const float FracX = SrcX - X0;
			const float FracY = SrcY - Y0;

			const float H00 = InHeightMap.Heights[Y0 * InHeightMap.Width + X0];
			const float H10 = InHeightMap.Heights[Y0 * InHeightMap.Width + X1];
			const float H01 = InHeightMap.Heights[Y1 * InHeightMap.Width + X0];
			const float H11 = InHeightMap.Heights[Y1 * InHeightMap.Width + X1];

			const float H0 = FMath::Lerp(H00, H10, FracX);
			const float H1 = FMath::Lerp(H01, H11, FracX);
			const float NormHeight = FMath::Lerp(H0, H1, FracY);

			const float Height_m = InHeightMap.MinElevation_m + NormHeight * ElevRange;
			const float Z_cm = Height_m * 100.0f * ZScale;

			OutVertices.Add(FVector(X * StepSize * 100.0f, Y * StepSize * 100.0f, Z_cm));
			OutColors.Add(GetColorByHeight(NormHeight));
			OutNormals.Add(FVector::UpVector);
		}
	}

	for (int32 Y = 0; Y < InResolution - 1; ++Y)
	{
		for (int32 X = 0; X < InResolution - 1; ++X)
		{
			const int32 I0 = Y * InResolution + X;
			const int32 I1 = I0 + 1;
			const int32 I2 = I0 + InResolution;
			const int32 I3 = I2 + 1;

			OutTriangles.Add(I0);
			OutTriangles.Add(I2);
			OutTriangles.Add(I1);

			OutTriangles.Add(I1);
			OutTriangles.Add(I2);
			OutTriangles.Add(I3);
		}
	}

	for (int32 i = 0; i < OutNormals.Num(); ++i)
	{
		OutNormals[i] = FVector::ZeroVector;
	}

	for (int32 t = 0; t < OutTriangles.Num(); t += 3)
	{
		const int32 A = OutTriangles[t];
		const int32 B = OutTriangles[t + 1];
		const int32 C = OutTriangles[t + 2];

		const FVector Normal = FVector::CrossProduct(
			OutVertices[B] - OutVertices[A],
			OutVertices[C] - OutVertices[A]).GetSafeNormal();

		OutNormals[A] += Normal;
		OutNormals[B] += Normal;
		OutNormals[C] += Normal;
	}

	for (FVector& N : OutNormals)
	{
		N = N.GetSafeNormal();
	}
}

	if (OldLocation.Equals(NewLocation, 1.0f))

		return;
	}

	SetActorLocation(NewLocation);

	UE_LOG(LogTemp, Log,
		TEXT("[RailEndpoint|%s] Move: (%.0f, %.0f, %.0f) → (%.0f, %.0f, %.0f). ")
		TEXT("TerrainZ=%.0f, Offset=%.0f."),
		bIsStart ? TEXT("Start") : TEXT("End"),
		OldLocation.X, OldLocation.Y, OldLocation.Z,
		NewLocation.X, NewLocation.Y, NewLocation.Z,
		InTerrainHitLocation.Z, TerrainZOffset_cm);

	if (bBroadcastEvent)
	{
		OnEndpointMoved.Broadcast(this);
	}
}

void ARailEndpoint::SetHighlighted(bool bHighlighted)
{
	if (!DynamicMaterial)
	{
		ApplyVisuals();
	}

	if (!DynamicMaterial) return;

	const float Emissive = bHighlighted ? HighlightEmissiveBoost : 0.0f;
	DynamicMaterial->SetScalarParameterValue(TEXT("EmissiveBoost"), Emissive);

	UE_LOG(LogTemp, Verbose,
		TEXT("[RailEndpoint|%s] Highlight=%s, EmissiveBoost=%.2f."),
		bIsStart ? TEXT("Start") : TEXT("End"),
		bHighlighted ? TEXT("on") : TEXT("off"),
		Emissive);
}

void ARailEndpoint::SetViewScale(float InScale)
{
	const float NewScale = FMath::Clamp(InScale, MinViewScale, MaxViewScale);

	if (FMath::IsNearlyEqual(NewScale, ViewScale, 0.01f))
	{
		return;
	}

	ViewScale = NewScale;
	SphereRadius_cm = BaseSphereRadius_cm * ViewScale;
	ApplyVisuals();

	UE_LOG(LogTemp, Log,
		TEXT("[RailEndpoint|%s] SetViewScale: %.2f → radius=%.0f см."),
		bIsStart ? TEXT("Start") : TEXT("End"),
		ViewScale, SphereRadius_cm);
}

void ARailEndpoint::ApplyVisuals()
{
	if (CollisionSphere)
	{
		CollisionSphere->SetSphereRadius(SphereRadius_cm, /*bUpdateOverlaps=*/false);
	}

	if (VisualMesh)
	{
		const float Scale = SphereRadius_cm / 50.0f;
		VisualMesh->SetRelativeScale3D(FVector(Scale));
	}

	if (VisualMesh && EndpointMaterial)
	{
		if (!DynamicMaterial || DynamicMaterial->Parent != EndpointMaterial)

			DynamicMaterial->SetScalarParameterValue(TEXT("EmissiveBoost"), 0.0f);

			VisualMesh->SetMaterial(0, DynamicMaterial);
		}
	}
}

void ARailOptimaPlayerController::PlayerTick(float DeltaTime)

ARailEndpoint* ARailOptimaPlayerController::PickEndpointUnderCursor() const
{
	FVector WorldOrigin, WorldDir;
	if (!DeprojectMousePositionToWorld(WorldOrigin, WorldDir)) return nullptr;

	const FVector End = WorldOrigin + WorldDir * 100'000'000.0f;

	FCollisionObjectQueryParams ObjParams;
	ObjParams.AddObjectTypesToQuery(ECC_WorldDynamic);

	FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(RailPickEndpoint), /*bTraceComplex=*/false);
	QueryParams.AddIgnoredActor(GetPawn());

	TArray<FHitResult> Hits;
	GetWorld()->SweepMultiByObjectType(
		Hits, WorldOrigin, End, FQuat::Identity,
		ObjParams, FCollisionShape::MakeSphere(EndpointPickRadius_cm), QueryParams);

	for (const FHitResult& Hit : Hits)
	{
		if (ARailEndpoint* Endpoint = Cast<ARailEndpoint>(Hit.GetActor()))
		{
			return Endpoint;
		}
	}
	return nullptr;
}

bool ARailOptimaPlayerController::TraceTerrainUnderCursor(FVector& OutHitLocation) const
{
	FVector WorldOrigin, WorldDir;
	if (!DeprojectMousePositionToWorld(WorldOrigin, WorldDir)) return false;

	const FVector End = WorldOrigin + WorldDir * 100'000'000.0f;

	FCollisionObjectQueryParams ObjParams;
	ObjParams.AddObjectTypesToQuery(ECC_WorldStatic);

	FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(RailTerrainCursorTrace), /*bTraceComplex=*/true);
	QueryParams.AddIgnoredActor(GetPawn());

	FHitResult Hit;
	const bool bHit = GetWorld()->LineTraceSingleByObjectType(
		Hit, WorldOrigin, End, ObjParams, QueryParams);

	if (const_cast<ARailOptimaPlayerController*>(this)->TraceFrameCounter % 15 == 0)
	{
		if (bHit)
		{
			UE_LOG(LogTemp, Log,
				TEXT("[RailPC] Трейс рельефа: HIT, actor='%s', impact=(%.0f, %.0f, %.0f)."),
				*GetNameSafe(Hit.GetActor()),
				Hit.ImpactPoint.X, Hit.ImpactPoint.Y, Hit.ImpactPoint.Z);
		}
		else
		{
			UE_LOG(LogTemp, Warning,
				TEXT("[RailPC] Трейс рельефа: MISS. Проверь complex collision у ProceduralTerrain."));
		}
	}

	if (!bHit) return false;

	OutHitLocation = Hit.ImpactPoint;
	return true;
}

void ARailSpectatorPawn::SetCameraMode(ERailCameraMode NewMode)
{
	if (CameraMode == NewMode && CameraBoom) return;

	const ERailCameraMode OldMode = CameraMode;

	// 1. Запомнить позицию камеры ДО смены режима.
	const FVector CameraOldLocation = Camera ? Camera->GetComponentLocation() : GetActorLocation();

	UE_LOG(LogTemp, Log,
		TEXT("[RailSpectator] Tab: %s → %s. Камера сейчас в (%.0f, %.0f, %.0f)."),
		OldMode == ERailCameraMode::FreeFly ? TEXT("FreeFly") : TEXT("TopDown"),
		NewMode == ERailCameraMode::FreeFly ? TEXT("FreeFly") : TEXT("TopDown"),
		CameraOldLocation.X, CameraOldLocation.Y, CameraOldLocation.Z);

	// 2. Сменить режим и применить настройки boom'а.
	CameraMode = NewMode;
	ApplyCameraMode();

	// 3. Позиционирование павна.
	FVector NewPawnLocation;

	if (NewMode == ERailCameraMode::FreeFly)
	{
		// В FreeFly ставим павн туда, где сейчас камера.
		// Никакого телепорта — камера остаётся на месте.
		NewPawnLocation = CameraOldLocation;

		// Установить ControlRotation так, чтобы камера смотрела туда же,
		// куда смотрела в TopDown (вниз). Иначе она повернётся горизонтально.
		if (APlayerController* PC = Cast<APlayerController>(GetController()))
		{
			PC->SetControlRotation(FRotator(TopDownPitch, 0.0f, 0.0f));
		}

		SetActorRotation(FRotator::ZeroRotator);
	}
	else  // TopDown
	{
		// Павн на "земле" под старой камерой (XY сохраняем, Z = 0).
		NewPawnLocation = FVector(CameraOldLocation.X, CameraOldLocation.Y, 0.0f);
		SetActorRotation(FRotator::ZeroRotator);
	}

	SetActorLocation(NewPawnLocation);

	UE_LOG(LogTemp, Log,
		TEXT("[RailSpectator] Pawn переставлен: (%.0f, %.0f, %.0f)."),
		NewPawnLocation.X, NewPawnLocation.Y, NewPawnLocation.Z);

	// 4. Сброс накопленных осей и отпускание RMB.
	if (bCameraLookActive)
	{
		SetCameraLookActive(false);
	}

	MoveInput = FVector2D::ZeroVector;
	MoveUpInput = 0.0f;
	LookInput = FVector2D::ZeroVector;
	ZoomInput = 0.0f;

	LogState(TEXT("SetCameraMode"));
}

bool ARailSpectatorPawn::CenterOnWorld(float InMapSizeX_cm, float InMapSizeY_cm, float InMapSizeZ_cm)
{
	if (!CameraBoom) return false;

	const FVector Center(InMapSizeX_cm * 0.5f, InMapSizeY_cm * 0.5f, 0.0f);
	SetActorLocation(Center);

	const float HalfMax = FMath::Max(InMapSizeX_cm, InMapSizeY_cm) * 0.5f;
	TopDownHeight_cm = FMath::Clamp(HalfMax * 1.2f, TopDownMinHeight_cm, TopDownMaxHeight_cm);

	ApplyCameraMode();

	UE_LOG(LogTemp, Log,
		TEXT("[RailSpectator] CenterOnWorld: pawn в (%.0f, %.0f, %.0f), TopDownHeight=%.0f см."),
		Center.X, Center.Y, Center.Z, TopDownHeight_cm);

	return true;
}

void ARailSpectatorPawn::OnZoom(const FInputActionValue& Value)
{
	if (CameraMode != ERailCameraMode::TopDown) return;

	const float Raw = Value.Get<float>();
	if (FMath::IsNearlyZero(Raw)) return;

	const float OldHeight = TopDownHeight_cm;
	TopDownHeight_cm *= (Raw > 0.0f) ? (1.0f / ZoomStep) : ZoomStep;
	TopDownHeight_cm = FMath::Clamp(TopDownHeight_cm, TopDownMinHeight_cm, TopDownMaxHeight_cm);

	if (!FMath::IsNearlyEqual(OldHeight, TopDownHeight_cm, 1.0f))
	{
		CameraBoom->TargetArmLength = TopDownHeight_cm;
		UE_LOG(LogTemp, Log, TEXT("[RailSpectator] Zoom: height %.0f → %.0f см."),
			OldHeight, TopDownHeight_cm);

void ARailSpectatorPawn::HandleMovement(float DeltaSeconds)
{
	const bool bHasMove2D = !MoveInput.IsNearlyZero();
	const bool bHasMoveUp = !FMath::IsNearlyZero(MoveUpInput);
	if (!bHasMove2D && !bHasMoveUp) return;

	// Boost работает в ОБОИХ режимах.
	const float Boost = bBoostHeld ? BoostMultiplier : 1.0f;

	if (CameraMode == ERailCameraMode::FreeFly)
	{
		const FRotator ControlRot = GetControlRotation();
		const FVector Forward = FRotationMatrix(ControlRot).GetUnitAxis(EAxis::X);
		const FVector Right = FRotationMatrix(ControlRot).GetUnitAxis(EAxis::Y);

		const float Speed = FreeFlySpeed_cmps * Boost;

		FVector Delta = (Forward * MoveInput.X + Right * MoveInput.Y) * Speed * DeltaSeconds;
		Delta.Z += MoveUpInput * VerticalSpeed_cmps * Boost * DeltaSeconds;

		AddActorWorldOffset(Delta, false);
	}
	else
	{
		// TopDown: панорамирование по мировой XY (W → +X, D → +Y),
		// Space/Ctrl по мировой Z. Boost ускоряет и то, и то.
		const FVector Delta(
			MoveInput.X * TopDownPanSpeed_cmps * Boost * DeltaSeconds,
			MoveInput.Y * TopDownPanSpeed_cmps * Boost * DeltaSeconds,
			MoveUpInput * VerticalSpeed_cmps * Boost * DeltaSeconds);

		AddActorWorldOffset(Delta, false);
	}
}

		TEXT("[RailSpectator] %s: mode=%s, pos=(%.0f, %.0f, %.0f), boomLength=%.0f см, ")
		TEXT("boomRot=(P=%.1f, Y=%.1f), ctrlRot=(P=%.1f, Y=%.1f)."),
		*InContext,
		(CameraMode == ERailCameraMode::FreeFly) ? TEXT("FreeFly") : TEXT("TopDown"),
		Loc.X, Loc.Y, Loc.Z,
		CameraBoom ? CameraBoom->TargetArmLength : -1.0f,
		CameraBoom ? CameraBoom->GetRelativeRotation().Pitch : 0.0f,
		CameraBoom ? CameraBoom->GetRelativeRotation().Yaw : 0.0f,
		ControlRot.Pitch, ControlRot.Yaw);
}

void APathRenderer::RenderPath(const FRailGridData& InGrid, const FRailPathResult& InPath)

void APathRenderer::RenderColoredPath(
	const FRailGridData& InGrid,
	const FRailPathResult& InPath,
	const TArray<float>& InNodeValues,
	float InLowValue,
	float InHighValue)
{
	if (!InPath.bFound || InPath.NodeIndices.Num() < 2)
	{
		ClearPath();
		return;
	}

	if (InNodeValues.Num() != InPath.NodeIndices.Num())
	{
		UE_LOG(LogTemp, Error,
			TEXT("[PathRenderer] Длина InNodeValues (%d) != числа узлов пути (%d)."),
			InNodeValues.Num(), InPath.NodeIndices.Num());
		return;
	}

	const float Range = FMath::Max(InHighValue - InLowValue, KINDA_SMALL_NUMBER);

	TArray<FVector> RawPoints;
	TArray<FLinearColor> RawColors;
	RawPoints.Reserve(InPath.NodeIndices.Num());
	RawColors.Reserve(InPath.NodeIndices.Num());

	for (int32 i = 0; i < InPath.NodeIndices.Num(); ++i)
	{
		const int32 NodeIndex = InPath.NodeIndices[i];
		if (!InGrid.Nodes.IsValidIndex(NodeIndex)) continue;

		const FRailGridNode& Node = InGrid.Nodes[NodeIndex];
		RawPoints.Add(GridNodeToWorld(Node, InGrid.StepSize_m));

		return;

	TArray<FLinearColor> FinalColors;

	{
		SmoothPolylineCatmullRom(

			TEXT("[PathRenderer] Сглаживание: %d → %d точек (Catmull-Rom, samples=%d)."),
			RawPoints.Num(), FinalPoints.Num(), SamplesPerSegment);
	}
	else
	{
		FinalPoints = RawPoints;
		FinalColors = RawColors;
		UE_LOG(LogTemp, Log, TEXT("[PathRenderer] Сглаживание отключено."));
	}

void APathRenderer::SmoothPolylineCatmullRom(
	const TArray<FVector>& InPoints,
	const TArray<FLinearColor>& InColors,
	int32 InSamplesPerSegment,
	TArray<FVector>& OutPoints,
	TArray<FLinearColor>& OutColors)
{
	const int32 N = InPoints.Num();
	if (N < 2 || InSamplesPerSegment < 1)
	{
		OutPoints = InPoints;
		OutColors = InColors;
		return;
	}

	OutPoints.Reset();
	OutColors.Reset();
	OutPoints.Reserve((N - 1) * InSamplesPerSegment + 1);
	OutColors.Reserve((N - 1) * InSamplesPerSegment + 1);

	auto GetPoint = [&](int32 Index) -> FVector
		{
			return InPoints[FMath::Clamp(Index, 0, N - 1)];
		};
	auto GetColor = [&](int32 Index) -> FLinearColor
		{
			return InColors[FMath::Clamp(Index, 0, N - 1)];
		};

	for (int32 i = 0; i < N - 1; ++i)
	{
		const FVector P0 = GetPoint(i - 1);
		const FVector P1 = GetPoint(i);
		const FVector P2 = GetPoint(i + 1);
		const FVector P3 = GetPoint(i + 2);

		const FLinearColor C1 = GetColor(i);
		const FLinearColor C2 = GetColor(i + 1);

		for (int32 s = 0; s < InSamplesPerSegment; ++s)
		{
			const float t = static_cast<float>(s) / static_cast<float>(InSamplesPerSegment);
			const float t2 = t * t;
			const float t3 = t2 * t;

			const FVector P =
				0.5f * (
					(2.0f * P1) +
					(-P0 + P2) * t +
					(2.0f * P0 - 5.0f * P1 + 4.0f * P2 - P3) * t2 +
					(-P0 + 3.0f * P1 - 3.0f * P2 + P3) * t3);

			OutPoints.Add(P);
			OutColors.Add(FMath::Lerp(C1, C2, t));
		}
	}

	OutPoints.Add(InPoints.Last());
	OutColors.Add(InColors.Last());
}

void APathRenderer::BuildRibbon(
	const TArray<FVector>& InPoints,
	const TArray<FLinearColor>& InColors,
	float InWidth_cm,
	float InHeight_cm)
{
	const int32 NumPoints = InPoints.Num();
	if (NumPoints < 2) return;

	const float HalfWidth = InWidth_cm * 0.5f;

	TArray<FVector2D> Directions;
	Directions.SetNumUninitialized(NumPoints);

	auto SegmentDir2D = [](const FVector& A, const FVector& B) -> FVector2D
		{
			FVector2D D(B.X - A.X, B.Y - A.Y);
			const float Len = D.Size();
			return (Len > KINDA_SMALL_NUMBER) ? (D / Len) : FVector2D(1.0f, 0.0f);
		};

	Directions[0] = SegmentDir2D(InPoints[0], InPoints[1]);
	Directions[NumPoints - 1] = SegmentDir2D(InPoints[NumPoints - 2], InPoints[NumPoints - 1]);

	for (int32 i = 1; i < NumPoints - 1; ++i)
	{
		const FVector2D Prev = SegmentDir2D(InPoints[i - 1], InPoints[i]);
		const FVector2D Next = SegmentDir2D(InPoints[i], InPoints[i + 1]);
		FVector2D Avg = Prev + Next;
		const float Len = Avg.Size();
		Directions[i] = (Len > KINDA_SMALL_NUMBER) ? (Avg / Len) : Next;
	}

	TArray<FVector>          Vertices;
	TArray<FLinearColor>     VertColors;
	TArray<FVector>          Normals;
	TArray<FVector2D>        UVs;
	TArray<FProcMeshTangent> Tangents;

	const int32 NumVertices = NumPoints * 4;
	Vertices.Reserve(NumVertices);
	VertColors.Reserve(NumVertices);
	Normals.Reserve(NumVertices);
	UVs.Reserve(NumVertices);
	Tangents.Reserve(NumVertices);

	for (int32 i = 0; i < NumPoints; ++i)
	{
		const FVector& P = InPoints[i];
		const FVector2D& Dir = Directions[i];
		const FVector2D Perp(-Dir.Y, Dir.X);
		const FVector Lateral(Perp.X * HalfWidth, Perp.Y * HalfWidth, 0.0f);

		const FVector L = P - Lateral;
		const FVector R = P + Lateral;

		const FVector LTop(L.X, L.Y, L.Z + InHeight_cm);
		const FVector RTop(R.X, R.Y, R.Z + InHeight_cm);

		const FProcMeshTangent Tangent(FVector(Dir.X, Dir.Y, 0.0f), false);

		Vertices.Add(LTop);
		VertColors.Add(InColors[i]);
		Normals.Add(FVector::UpVector);
		UVs.Add(FVector2D(0.0f, static_cast<float>(i)));
		Tangents.Add(Tangent);

		Vertices.Add(RTop);
		VertColors.Add(InColors[i]);
		Normals.Add(FVector::UpVector);
		UVs.Add(FVector2D(1.0f, static_cast<float>(i)));
		Tangents.Add(Tangent);

		Vertices.Add(L);
		VertColors.Add(InColors[i]);
		Normals.Add(FVector(-Perp.X, -Perp.Y, 0.0f));
		UVs.Add(FVector2D(0.0f, static_cast<float>(i)));
		Tangents.Add(Tangent);

		Vertices.Add(R);
		VertColors.Add(InColors[i]);
		Normals.Add(FVector(Perp.X, Perp.Y, 0.0f));
		UVs.Add(FVector2D(1.0f, static_cast<float>(i)));
		Tangents.Add(Tangent);
	}

	TArray<int32> Triangles;
	Triangles.Reserve((NumPoints - 1) * 18);

	for (int32 i = 0; i < NumPoints - 1; ++i)
	{
		const int32 LTop0 = i * 4 + 0;
		const int32 RTop0 = i * 4 + 1;
		const int32 LBot0 = i * 4 + 2;
		const int32 RBot0 = i * 4 + 3;

		const int32 LTop1 = (i + 1) * 4 + 0;
		const int32 RTop1 = (i + 1) * 4 + 1;
		const int32 LBot1 = (i + 1) * 4 + 2;
		const int32 RBot1 = (i + 1) * 4 + 3;

		Triangles.Add(LTop0); Triangles.Add(RTop0); Triangles.Add(LTop1);
		Triangles.Add(RTop0); Triangles.Add(RTop1); Triangles.Add(LTop1);

		Triangles.Add(LTop0); Triangles.Add(LBot0); Triangles.Add(LTop1);
		Triangles.Add(LBot0); Triangles.Add(LBot1); Triangles.Add(LTop1);

		Triangles.Add(RTop0); Triangles.Add(RTop1); Triangles.Add(RBot0);
		Triangles.Add(RBot0); Triangles.Add(RTop1); Triangles.Add(RBot1);
	}

	PathMesh->ClearAllMeshSections();
	PathMesh->CreateMeshSection_LinearColor(
		0,
		Vertices,
		Triangles,
		Normals,
		UVs,
		VertColors,
		Tangents,
		/*bCreateCollision=*/ false,
		/*bSRGBConversion=*/ true);

	if (PathMaterial)
	{
		PathMesh->SetMaterial(0, PathMaterial);
	}

	if (bLogVertexColors && VertColors.Num() > 0)
	{
		const FLinearColor& First = VertColors[0];
		const FLinearColor& Last = VertColors[VertColors.Num() - 1];

		UE_LOG(LogTemp, Log,
			TEXT("[PathRenderer] Лента: %d точек, %d вершин, %d треугольников. ")
			TEXT("Первый (R=%.2f G=%.2f B=%.2f), последний (R=%.2f G=%.2f B=%.2f)."),
			NumPoints, Vertices.Num(), Triangles.Num() / 3,
			First.R, First.G, First.B,
			Last.R, Last.G, Last.B);
	}
}

bool ARoadRenderer::LoadAndBuild(const FString& InGeoJsonPath)
{
	if (!Terrain)
	{
		UE_LOG(LogTemp, Error, TEXT("[RoadRenderer] Terrain == nullptr."));
		return false;
	}

	FString RawJson;
	if (!FFileHelper::LoadFileToString(RawJson, *InGeoJsonPath))
	{
		UE_LOG(LogTemp, Error,
			TEXT("[RoadRenderer] Не удалось прочитать: %s"), *InGeoJsonPath);
		return false;
	}

	TArray<TArray<FVector2D>> Lines;
	if (!ParseGeoJson(RawJson, Lines))
	{
		UE_LOG(LogTemp, Error, TEXT("[RoadRenderer] Не удалось распарсить GeoJSON."));
		return false;
	}

	if (Lines.Num() == 0)
	{
		UE_LOG(LogTemp, Warning, TEXT("[RoadRenderer] Нет LineString."));
		return false;
	}

	const FRailHeightMapData& MapData = Terrain->GetHeightMapData();

	UE_LOG(LogTemp, Log,
		TEXT("[RoadRenderer] MapData: OriginLat=%.6f, OriginLon=%.6f, ")
		TEXT("PixelSize=%.2f, Width=%d, ZOffset=%.0f см, AutoZOffset=%s, ")
		TEXT("MinSpacing=%.0f м, MaxVertsPerSection=%d."),
		MapData.OriginLat, MapData.OriginLon,
		MapData.PixelSize_m, MapData.Width,
		ZOffset_cm,
		bAutoZOffset ? TEXT("true") : TEXT("false"),
		MinPointSpacing_m, MaxVertsPerSection);

	if (MapData.Width <= 0)
	{
		UE_LOG(LogTemp, Error,
			TEXT("[RoadRenderer] Рельеф ещё не построен (Width=%d). "),
			MapData.Width);
		return false;
	}

	// Диагностика первой точки.
	if (Lines.Num() > 0 && Lines[0].Num() > 0)
	{
		const FVector2D FirstWorld = GeoToWorld(Lines[0][0], MapData);
		float FirstZ = 0.0f;
		const bool bHit = Terrain->SampleTerrainZ_cm(FirstWorld, FirstZ);
		UE_LOG(LogTemp, Warning,
			TEXT("[RoadRenderer] Первая точка GeoJSON: LatLon=(%.6f, %.6f) → ")
			TEXT("World=(%.0f, %.0f), Z=%.0f см (ok=%s)"),
			Lines[0][0].X, Lines[0][0].Y,
			FirstWorld.X, FirstWorld.Y,
			FirstZ, bHit ? TEXT("yes") : TEXT("no"));
	}

	const FLinearColor Color = (FeatureType == ERoadFeatureType::River)
		? RiverColor : RoadColor;

	if (RoadMesh) RoadMesh->ClearAllMeshSections();
	SectionBuffer.Reset();
	CurrentSectionIndex = 0;
	TotalVertsWritten = 0;
	TotalTrisWritten = 0;
	SectionsCreated = 0;

	const float MinSpacingCm = FMath::Max(MinPointSpacing_m, 0.0f) * 100.0f;

	int32 TotalLines = 0;
	int32 TotalPointsRaw = 0;
	int32 TotalPointsSimplified = 0;
	int32 TotalPointsAfterZ = 0;
	int32 TraceHits = 0;
	int32 TraceMisses = 0;

	for (const TArray<FVector2D>& Line : Lines)
	{
		if (Line.Num() < 2) continue;

		TArray<FVector2D> WorldXY;
		WorldXY.Reserve(Line.Num());
		for (const FVector2D& LatLon : Line)
		{
			WorldXY.Add(GeoToWorld(LatLon, MapData));
		}
		TotalPointsRaw += WorldXY.Num();

		TArray<FVector2D> Simplified;
		SimplifyLine(WorldXY, MinSpacingCm, Simplified);
		if (Simplified.Num() < 2) continue;

		TotalPointsSimplified += Simplified.Num();

		TArray<float> Zs;
		Zs.SetNumUninitialized(Simplified.Num());
		for (int32 i = 0; i < Simplified.Num(); ++i)
		{
			float Z = 0.0f;
			if (Terrain->SampleTerrainZ_cm(Simplified[i], Z))
			{
				++TraceHits;
			}
			else
			{
				++TraceMisses;
				Z = GetTerrainZFromMap(Simplified[i], MapData);
			}
			Zs[i] = Z;
		}

		SmoothZ(Zs, ZSmoothingRadius);

		TArray<FVector> WorldPoints;
		WorldPoints.Reserve(Simplified.Num());
		for (int32 i = 0; i < Simplified.Num(); ++i)
		{
			WorldPoints.Add(FVector(
				Simplified[i].X,
				Simplified[i].Y,
				Zs[i] + ZOffset_cm));
		}

		TotalPointsAfterZ += WorldPoints.Num();

		AppendRibbon(WorldPoints, Color);
		++TotalLines;
	}

	FlushSection();

	if (RoadMaterial)
	{
		RoadMesh->SetMaterial(0, RoadMaterial);
	}

	if (bLogStats)
	{
		UE_LOG(LogTemp, Log,
			TEXT("[RoadRenderer] Построено: линий=%d, точек raw=%d → simplified=%d → ")
			TEXT("после Z=%d, вершин=%d, треугольников=%d, секций=%d, ")
			TEXT("SampleTerrain hit=%d / miss=%d."),
			TotalLines,
			TotalPointsRaw, TotalPointsSimplified, TotalPointsAfterZ,
			TotalVertsWritten, TotalTrisWritten, SectionsCreated,
			TraceHits, TraceMisses);
	}

	return true;
}

// ─────────────────────────────────────────────────────────────────
//  Парсинг GeoJSON
// ─────────────────────────────────────────────────────────────────

bool ARoadRenderer::ParseGeoJson(const FString& InRawJson,
	TArray<TArray<FVector2D>>& OutLines) const
{
	OutLines.Reset();

	TSharedPtr<FJsonObject> Root;
	const TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(InRawJson);

	if (!FJsonSerializer::Deserialize(Reader, Root) || !Root.IsValid())
		return false;

	const TArray<TSharedPtr<FJsonValue>>* FeaturesArray = nullptr;
	if (!Root->TryGetArrayField(TEXT("features"), FeaturesArray) || !FeaturesArray)
		return false;

	for (const TSharedPtr<FJsonValue>& FeatureValue : *FeaturesArray)
	{
		const TSharedPtr<FJsonObject>* FeatureObj = nullptr;
		if (!FeatureValue->TryGetObject(FeatureObj) || !FeatureObj) continue;

		const TSharedPtr<FJsonObject>* GeometryObj = nullptr;
		if (!(*FeatureObj)->TryGetObjectField(TEXT("geometry"), GeometryObj) || !GeometryObj) continue;

		FString GeomType;
		if (!(*GeometryObj)->TryGetStringField(TEXT("type"), GeomType)) continue;

		if (GeomType == TEXT("LineString"))
		{
			TArray<FVector2D> Line;
			const TArray<TSharedPtr<FJsonValue>>* CoordsArray = nullptr;
			if ((*GeometryObj)->TryGetArrayField(TEXT("coordinates"), CoordsArray) && CoordsArray)
			{
				for (const TSharedPtr<FJsonValue>& CoordValue : *CoordsArray)
				{
					const TArray<TSharedPtr<FJsonValue>>* Pair = nullptr;
					if (!CoordValue->TryGetArray(Pair) || !Pair || Pair->Num() < 2) continue;
					const double Lon = (*Pair)[0]->AsNumber();
					const double Lat = (*Pair)[1]->AsNumber();
					Line.Add(FVector2D(Lat, Lon));
				}
			}
			if (Line.Num() >= 2) OutLines.Add(MoveTemp(Line));
		}
		else if (GeomType == TEXT("MultiLineString"))
		{
			const TArray<TSharedPtr<FJsonValue>>* MultiCoords = nullptr;
			if (!(*GeometryObj)->TryGetArrayField(TEXT("coordinates"), MultiCoords) || !MultiCoords) continue;

			for (const TSharedPtr<FJsonValue>& LineValue : *MultiCoords)
			{
				const TArray<TSharedPtr<FJsonValue>>* CoordsArray = nullptr;
				if (!LineValue->TryGetArray(CoordsArray) || !CoordsArray) continue;

				TArray<FVector2D> Line;
				for (const TSharedPtr<FJsonValue>& CoordValue : *CoordsArray)
				{
					const TArray<TSharedPtr<FJsonValue>>* Pair = nullptr;
					if (!CoordValue->TryGetArray(Pair) || !Pair || Pair->Num() < 2) continue;
					const double Lon = (*Pair)[0]->AsNumber();
					const double Lat = (*Pair)[1]->AsNumber();
					Line.Add(FVector2D(Lat, Lon));
				}
				if (Line.Num() >= 2) OutLines.Add(MoveTemp(Line));
			}
		}
	}

	return true;
}

// ─────────────────────────────────────────────────────────────────
//  Прореживание
// ─────────────────────────────────────────────────────────────────

void ARoadRenderer::SimplifyLine(const TArray<FVector2D>& InLine,
	float InMinSpacingCm,
	TArray<FVector2D>& OutLine)
{
	OutLine.Reset();
	if (InLine.Num() == 0) return;

	OutLine.Add(InLine[0]);

	if (InMinSpacingCm <= KINDA_SMALL_NUMBER)
	{
		for (int32 i = 1; i < InLine.Num(); ++i)
		{
			OutLine.Add(InLine[i]);
		}
		return;
	}

	const float SpacingSq = InMinSpacingCm * InMinSpacingCm;

	for (int32 i = 1; i < InLine.Num(); ++i)
	{
		const FVector2D& Last = OutLine.Last();
		const FVector2D& Curr = InLine[i];
		const float DistSq = FVector2D::DistSquared(Last, Curr);

		if (DistSq >= SpacingSq)
		{
			OutLine.Add(Curr);
		}
	}

	const FVector2D& LastOrig = InLine.Last();
	if (!OutLine.Last().Equals(LastOrig, 1.0f))
	{
		OutLine.Add(LastOrig);
	}
}

// ─────────────────────────────────────────────────────────────────
//  Координаты — с инверсией Y
// ─────────────────────────────────────────────────────────────────

FVector2D ARoadRenderer::GeoToWorld(const FVector2D& InLatLon,
	const FRailHeightMapData& MapData) const
{
	const double LatMetersPerDegree = 111320.0;
	const double CenterLat = MapData.OriginLat + 0.5;
	const double LonMetersPerDegree =
		111320.0 * FMath::Cos(FMath::DegreesToRadians(CenterLat));

	double MapSizeMeters = 0.0;
	if (Terrain)
	{
		MapSizeMeters = Terrain->GetMapSizeMeters();
	}
	else
	{
		MapSizeMeters = MapData.Width * MapData.PixelSize_m;
	}

	const double LatSpanDeg = MapSizeMeters / LatMetersPerDegree;
	const double LonSpanDeg = MapSizeMeters / LonMetersPerDegree;

	double NormX = (InLatLon.Y - MapData.OriginLon) / LonSpanDeg;

	// Y: инверсия. PNG строка 0 = север, Lat растёт к северу.
	double NormY = 1.0 - (InLatLon.X - MapData.OriginLat) / LatSpanDeg;

	if (bClampToMapBounds)
	{
		NormX = FMath::Clamp(NormX, 0.0, 1.0);
		NormY = FMath::Clamp(NormY, 0.0, 1.0);
	}

	return FVector2D(NormX * MapSizeMeters * 100.0,
		NormY * MapSizeMeters * 100.0);
}

// ─────────────────────────────────────────────────────────────────
//  Fallback — интерполяция по PNG
// ─────────────────────────────────────────────────────────────────

float ARoadRenderer::GetTerrainZFromMap(const FVector2D& InWorldXY,
	const FRailHeightMapData& MapData) const
{
	if (!Terrain) return 0.0f;

	const double MapSizeCm = Terrain->GetMapSizeMeters() * 100.0;
	if (MapSizeCm <= 0.0) return 0.0f;

	const double NormX = FMath::Clamp(InWorldXY.X / MapSizeCm, 0.0, 1.0);
	const double NormY = FMath::Clamp(InWorldXY.Y / MapSizeCm, 0.0, 1.0);

	const double PixelX = NormX * (MapData.Width - 1);
	const double PixelY = NormY * (MapData.Height - 1);

	const int32 X0 = FMath::Clamp(FMath::FloorToInt(PixelX), 0, MapData.Width - 1);
	const int32 Y0 = FMath::Clamp(FMath::FloorToInt(PixelY), 0, MapData.Height - 1);
	const int32 X1 = FMath::Clamp(X0 + 1, 0, MapData.Width - 1);
	const int32 Y1 = FMath::Clamp(Y0 + 1, 0, MapData.Height - 1);

	const float FracX = static_cast<float>(PixelX - X0);
	const float FracY = static_cast<float>(PixelY - Y0);

	auto Sample = [&](int32 X, int32 Y) -> float
		{
			return MapData.Heights[Y * MapData.Width + X];
		};

	const float H00 = Sample(X0, Y0);
	const float H10 = Sample(X1, Y0);
	const float H01 = Sample(X0, Y1);
	const float H11 = Sample(X1, Y1);

	const float H0 = FMath::Lerp(H00, H10, FracX);
	const float H1 = FMath::Lerp(H01, H11, FracX);
	const float NormH = FMath::Lerp(H0, H1, FracY);

	const float ElevRange = MapData.MaxElevation_m - MapData.MinElevation_m;
	const float Height_m = MapData.MinElevation_m + NormH * ElevRange;

	return Height_m * 100.0f * WorldZScale;
}

// ─────────────────────────────────────────────────────────────────
//  Сглаживание Z
// ─────────────────────────────────────────────────────────────────

void ARoadRenderer::SmoothZ(TArray<float>& InOutZ, int32 InRadius)
{
	const int32 N = InOutZ.Num();
	if (InRadius <= 0 || N < 3) return;

	TArray<float> Source = InOutZ;

	for (int32 i = 0; i < N; ++i)
	{
		const int32 Lo = FMath::Max(0, i - InRadius);
		const int32 Hi = FMath::Min(N - 1, i + InRadius);

		float Sum = 0.0f;
		int32 Count = 0;
		for (int32 j = Lo; j <= Hi; ++j)
		{
			Sum += Source[j];
			++Count;
		}
		InOutZ[i] = (Count > 0) ? (Sum / Count) : Source[i];
	}
}

// ─────────────────────────────────────────────────────────────────
//  Лента
// ─────────────────────────────────────────────────────────────────

void ARoadRenderer::AppendRibbon(const TArray<FVector>& InPoints,
	const FLinearColor& InColor)
{
	const int32 N = InPoints.Num();
	if (N < 2) return;

	const float HalfWidth = Width_cm * 0.5f;

	TArray<FVector2D> Directions;
	Directions.SetNumUninitialized(N);

	auto Seg2D = [](const FVector& A, const FVector& B) -> FVector2D
		{
			FVector2D D(B.X - A.X, B.Y - A.Y);
			const float L = D.Size();
			return (L > KINDA_SMALL_NUMBER) ? (D / L) : FVector2D(1.0f, 0.0f);
		};

	Directions[0] = Seg2D(InPoints[0], InPoints[1]);
	Directions[N - 1] = Seg2D(InPoints[N - 2], InPoints[N - 1]);

	for (int32 i = 1; i < N - 1; ++i)
	{
		const FVector2D P = Seg2D(InPoints[i - 1], InPoints[i]);
		const FVector2D Q = Seg2D(InPoints[i], InPoints[i + 1]);
		const FVector2D Avg = P + Q;
		const float L = Avg.Size();
		Directions[i] = (L > KINDA_SMALL_NUMBER) ? (Avg / L) : Q;
	}

	const int32 VertsNeeded = (N - 1) * 2;

	if (VertsNeeded > MaxVertsPerSection)
	{
		const int32 MaxPointsPerChunk = (MaxVertsPerSection / 2) + 1;

		for (int32 Start = 0; Start < N - 1; Start += MaxPointsPerChunk - 1)
		{
			const int32 End = FMath::Min(Start + MaxPointsPerChunk, N);
			const int32 ChunkSize = End - Start;
			if (ChunkSize < 2) break;

			if (SectionBuffer.NumVerts() + (ChunkSize - 1) * 2 > MaxVertsPerSection)
			{
				FlushSection();
			}

			TArray<FVector> ChunkPoints;
			TArray<FVector2D> ChunkDirs;
			ChunkPoints.Reserve(ChunkSize);
			ChunkDirs.Reserve(ChunkSize);
			for (int32 k = 0; k < ChunkSize; ++k)
			{
				ChunkPoints.Add(InPoints[Start + k]);
				ChunkDirs.Add(Directions[Start + k]);
			}

			const int32 BaseIndex = SectionBuffer.Vertices.Num();

			for (int32 i = 0; i < ChunkSize; ++i)
			{
				const FVector& P = ChunkPoints[i];
				const FVector2D& D = ChunkDirs[i];
				const FVector2D Perp(-D.Y, D.X);
				const FVector Lateral(Perp.X * HalfWidth, Perp.Y * HalfWidth, 0.0f);

				SectionBuffer.Vertices.Add(P - Lateral);
				SectionBuffer.Colors.Add(InColor);
				SectionBuffer.Normals.Add(FVector::UpVector);
				SectionBuffer.UVs.Add(FVector2D(0.0f, static_cast<float>(i)));
				SectionBuffer.Tangents.Add(FProcMeshTangent(FVector(D.X, D.Y, 0.0f), false));

				SectionBuffer.Vertices.Add(P + Lateral);
				SectionBuffer.Colors.Add(InColor);
				SectionBuffer.Normals.Add(FVector::UpVector);
				SectionBuffer.UVs.Add(FVector2D(1.0f, static_cast<float>(i)));
				SectionBuffer.Tangents.Add(FProcMeshTangent(FVector(D.X, D.Y, 0.0f), false));
			}

			for (int32 i = 0; i < ChunkSize - 1; ++i)
			{
				const int32 L0 = BaseIndex + i * 2 + 0;
				const int32 R0 = BaseIndex + i * 2 + 1;
				const int32 L1 = BaseIndex + (i + 1) * 2 + 0;
				const int32 R1 = BaseIndex + (i + 1) * 2 + 1;

				SectionBuffer.Triangles.Add(L0);
				SectionBuffer.Triangles.Add(R0);
				SectionBuffer.Triangles.Add(L1);
				SectionBuffer.Triangles.Add(R0);
				SectionBuffer.Triangles.Add(R1);
				SectionBuffer.Triangles.Add(L1);
			}
		}
		return;
	}

	if (SectionBuffer.NumVerts() + VertsNeeded > MaxVertsPerSection)
	{
		FlushSection();
	}

	const int32 BaseIndex = SectionBuffer.Vertices.Num();

	for (int32 i = 0; i < N; ++i)
	{
		const FVector& P = InPoints[i];
		const FVector2D& D = Directions[i];
		const FVector2D Perp(-D.Y, D.X);
		const FVector Lateral(Perp.X * HalfWidth, Perp.Y * HalfWidth, 0.0f);

		SectionBuffer.Vertices.Add(P - Lateral);
		SectionBuffer.Colors.Add(InColor);
		SectionBuffer.Normals.Add(FVector::UpVector);
		SectionBuffer.UVs.Add(FVector2D(0.0f, static_cast<float>(i)));
		SectionBuffer.Tangents.Add(FProcMeshTangent(FVector(D.X, D.Y, 0.0f), false));

		SectionBuffer.Vertices.Add(P + Lateral);
		SectionBuffer.Colors.Add(InColor);
		SectionBuffer.Normals.Add(FVector::UpVector);
		SectionBuffer.UVs.Add(FVector2D(1.0f, static_cast<float>(i)));
		SectionBuffer.Tangents.Add(FProcMeshTangent(FVector(D.X, D.Y, 0.0f), false));
	}

	for (int32 i = 0; i < N - 1; ++i)
	{
		const int32 L0 = BaseIndex + i * 2 + 0;
		const int32 R0 = BaseIndex + i * 2 + 1;
		const int32 L1 = BaseIndex + (i + 1) * 2 + 0;
		const int32 R1 = BaseIndex + (i + 1) * 2 + 1;

		SectionBuffer.Triangles.Add(L0);
		SectionBuffer.Triangles.Add(R0);
		SectionBuffer.Triangles.Add(L1);
		SectionBuffer.Triangles.Add(R0);
		SectionBuffer.Triangles.Add(R1);
		SectionBuffer.Triangles.Add(L1);
	}
}

void ARailOptimaGameMode::ApplySettings(
	float InSlopePenalty, float InTurnPenalty, float InStepSize_m, bool bInColorByCost)
{
	if (!CostCalculator)
	{
		UE_LOG(LogTemp, Warning, TEXT("[RailGM] ApplySettings: CostCalculator == nullptr."));
		return;
	}

	const bool bStepChanged =
		!FMath::IsNearlyEqual(RuntimeSettings.TargetStepSize_m, InStepSize_m, 1.0f);

	RuntimeSettings.SlopePenalty = FMath::Clamp(InSlopePenalty, 0.0f, 100.0f);
	RuntimeSettings.TurnPenalty = FMath::Clamp(InTurnPenalty, 0.0f, 100.0f);
	RuntimeSettings.TargetStepSize_m = FMath::Clamp(InStepSize_m, 10.0f, 1000.0f);
	RuntimeSettings.bColorPathByCost = bInColorByCost;

	CostCalculator->Parameters.SlopePenalty = RuntimeSettings.SlopePenalty;
	CostCalculator->Parameters.TurnPenalty = RuntimeSettings.TurnPenalty;

	if (bStepChanged)
	{
		bGridCached = false;
		CachedGrid = FRailGridData();
		UE_LOG(LogTemp, Log, TEXT("[RailGM] ApplySettings: шаг сетки изменён, кэш сброшен."));
	}

	UE_LOG(LogTemp, Log,
		TEXT("[RailGM] ApplySettings: Slope=%.3f, Turn=%.3f, Step=%.1f, ColorByCost=%s."),
		RuntimeSettings.SlopePenalty, RuntimeSettings.TurnPenalty,
		RuntimeSettings.TargetStepSize_m,
		RuntimeSettings.bColorPathByCost ? TEXT("true") : TEXT("false"));
}

void ARailOptimaGameMode::ApplyViewScaleFromCamera(ARailSpectatorPawn* InPawn)
{
	if (!InPawn)
	{
		return;
	}

	float TargetScale = 1.0f;

	if (InPawn->GetCameraMode() == ERailCameraMode::TopDown)
	{
		const float CurrentHeight = InPawn->GetTopDownHeight();

		if (!bViewScaleReferenceCaptured && CurrentHeight > 0.0f)
		{
			ReferenceTopDownHeight_cm = CurrentHeight;
			bViewScaleReferenceCaptured = true;

			UE_LOG(LogTemp, Log,
				TEXT("[RailGM] Захвачена референсная высота TopDown: %.0f см."),
				ReferenceTopDownHeight_cm);
		}

		if (ReferenceTopDownHeight_cm > 0.0f)
		{
			TargetScale = CurrentHeight / ReferenceTopDownHeight_cm;
		}
	}

	TargetScale = FMath::Clamp(TargetScale, MinViewScale, MaxViewScale);

	if (FMath::IsNearlyEqual(TargetScale, CurrentViewScale, 0.02f))
	{
		return;
	}

	CurrentViewScale = TargetScale;

	UE_LOG(LogTemp, Log,
		TEXT("[RailGM] ViewScale = %.2f (camera mode=%s)."),
		CurrentViewScale,
		(InPawn->GetCameraMode() == ERailCameraMode::FreeFly) ? TEXT("FreeFly") : TEXT("TopDown"));

	if (StartEndpoint) StartEndpoint->SetViewScale(CurrentViewScale);
	if (EndEndpoint)   EndEndpoint->SetViewScale(CurrentViewScale);

	if (PathRenderer)
	{
		PathRenderer->SetViewScale(CurrentViewScale);

		if (bGridCached && CachedGrid.Width > 1)
		{
			RecomputePath();
		}
	}
}

void ARailOptimaGameMode::SetupEndpoints()
{
	TArray<AActor*> Found;
	UGameplayStatics::GetAllActorsOfClass(GetWorld(), ARailEndpoint::StaticClass(), Found);

	for (AActor* Actor : Found)
	{
		ARailEndpoint* E = Cast<ARailEndpoint>(Actor);
		if (!E) continue;
		if (E->bIsStart && !StartEndpoint) StartEndpoint = E;
		else if (!E->bIsStart && !EndEndpoint) EndEndpoint = E;
	}

	FVector DefaultStartHit = FVector(0, 0, 0);
	FVector DefaultEndHit = FVector(0, 0, 0);

	if (Terrain)
	{
		const FRailHeightMapData& Map = Terrain->GetHeightMapData();
		const float SideX_cm = Map.Width * static_cast<float>(Map.PixelSize_m) * 100.0f;
		const float SideY_cm = Map.Height * static_cast<float>(Map.PixelSize_m) * 100.0f;
		const FVector TLoc = Terrain->GetActorLocation();
		const float M = AutoSpawnMargin_m * 100.0f;
		DefaultStartHit = FVector(TLoc.X + M, TLoc.Y + M, 0.0f);
		DefaultEndHit = FVector(TLoc.X + SideX_cm - M, TLoc.Y + SideY_cm - M, 0.0f);
	}

	FActorSpawnParameters SP;
	SP.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AlwaysSpawn;

	if (!StartEndpoint)
	{
		StartEndpoint = GetWorld()->SpawnActor<ARailEndpoint>(
			ARailEndpoint::StaticClass(), FVector(0, 0, 0), FRotator::ZeroRotator, SP);
		if (StartEndpoint)
		{
			StartEndpoint->bIsStart = true;
			StartEndpoint->EndpointColor = FLinearColor(0.1f, 0.9f, 0.1f, 1.0f);
			StartEndpoint->SetEndpointFromTerrainHit(DefaultStartHit, false);
			UE_LOG(LogTemp, Log, TEXT("[RailGM] StartEndpoint создан программно."));
		}
	}
	else
	{
		UE_LOG(LogTemp, Log, TEXT("[RailGM] StartEndpoint с уровня: '%s'."), *StartEndpoint->GetName());
	}

	if (!EndEndpoint)
	{
		EndEndpoint = GetWorld()->SpawnActor<ARailEndpoint>(
			ARailEndpoint::StaticClass(), FVector(0, 0, 0), FRotator::ZeroRotator, SP);
		if (EndEndpoint)
		{
			EndEndpoint->bIsStart = false;
			EndEndpoint->EndpointColor = FLinearColor(0.9f, 0.1f, 0.1f, 1.0f);
			EndEndpoint->SetEndpointFromTerrainHit(DefaultEndHit, false);
			UE_LOG(LogTemp, Log, TEXT("[RailGM] EndEndpoint создан программно."));
		}
	}
	else
	{
		UE_LOG(LogTemp, Log, TEXT("[RailGM] EndEndpoint с уровня: '%s'."), *EndEndpoint->GetName());
	}

	if (StartEndpoint) StartEndpoint->OnEndpointMoved.AddDynamic(this, &ARailOptimaGameMode::HandleEndpointMoved);
	if (EndEndpoint)   EndEndpoint->OnEndpointMoved.AddDynamic(this, &ARailOptimaGameMode::HandleEndpointMoved);

	UE_LOG(LogTemp, Log, TEXT("[RailGM] Маркеры готовы."));
}

// ─────────────────────────────────────────────────────────────────
//  Пересчёт
// ─────────────────────────────────────────────────────────────────

void ARailOptimaGameMode::RecomputePath()
{
	if (!PathFinder || !PathRenderer || !StartEndpoint || !EndEndpoint)
	{
		UE_LOG(LogTemp, Warning, TEXT("[RailGM] RecomputePath: не готова связка."));
		return;
	}

	if (!bGridCached)
	{
		if (Terrain && Terrain->GetHeightMapData().Heights.Num() > 0)
		{
			CachedGrid = BuildGridFromTerrain(*Terrain, RuntimeSettings.TargetStepSize_m);
		}
		else
		{
			CachedGrid = GenerateSyntheticGrid(336, RuntimeSettings.TargetStepSize_m);
		}
		bGridCached = CachedGrid.Width > 1 && CachedGrid.Height > 1;
	}

	if (!bGridCached)
	{
		UE_LOG(LogTemp, Error, TEXT("[RailGM] Сетка не построена."));
		return;
	}

	const int32 StartIdx = WorldToNodeIndex(CachedGrid, StartEndpoint->GetActorLocation());
	const int32 EndIdx = WorldToNodeIndex(CachedGrid, EndEndpoint->GetActorLocation());

	if (StartIdx == INDEX_NONE || EndIdx == INDEX_NONE || StartIdx == EndIdx)
	{
		UE_LOG(LogTemp, Warning, TEXT("[RailGM] Некорректные узлы."));
		PathRenderer->ClearPath();
		OnPathRecomputed.Broadcast(0.0, 0.0, 0.0, 0);
		return;
	}

	const double T0 = FPlatformTime::Seconds();
	const FRailPathResult Result = PathFinder->FindPath(CachedGrid, StartIdx, EndIdx);
	const double Ms = (FPlatformTime::Seconds() - T0) * 1000.0;

	if (!Result.bFound)
	{
		UE_LOG(LogTemp, Warning, TEXT("[RailGM] Путь не найден. %.1f мс, расширений=%d."),
			Ms, Result.ExpandedNodes);
		PathRenderer->ClearPath();
		OnPathRecomputed.Broadcast(0.0, 0.0, Ms, 0);
		return;
	}

	UE_LOG(LogTemp, Log,
		TEXT("[RailGM] Путь: длина=%.0f м, стоимость=%.1f ед., узлов=%d, рёбер=%d, время=%.1f мс."),
		Result.TotalLength_m, Result.TotalCost,
		Result.NodeIndices.Num(), Result.EdgeCosts.Num(), Ms);

	if (!RuntimeSettings.bColorPathByCost)
	{
		PathRenderer->RenderPath(CachedGrid, Result);
	}
	else
	{
		const int32 NumNodes = Result.NodeIndices.Num();
		TArray<float> NodeUnitCosts;
		NodeUnitCosts.Reserve(NumNodes);

		float MinU = TNumericLimits<float>::Max();
		float MaxU = TNumericLimits<float>::Lowest();

		for (int32 i = 0; i < NumNodes; ++i)
		{
			float Unit = 0.0f;
			if (i < Result.EdgeCosts.Num() && Result.EdgeLengths[i] > KINDA_SMALL_NUMBER)
			{
				Unit = Result.EdgeCosts[i] / Result.EdgeLengths[i];
			}
			else if (Result.EdgeCosts.Num() > 0 && Result.EdgeLengths.Last() > KINDA_SMALL_NUMBER)
			{
				Unit = Result.EdgeCosts.Last() / Result.EdgeLengths.Last();
			}
			NodeUnitCosts.Add(Unit);
			MinU = FMath::Min(MinU, Unit);
			MaxU = FMath::Max(MaxU, Unit);
		}

		if (FMath::IsNearlyEqual(MinU, MaxU)) MaxU = MinU + 1.0f;

		UE_LOG(LogTemp, Log,
			TEXT("[RailGM] Раскраска по стоимости: unit-cost min=%.3f, max=%.3f."),
			MinU, MaxU);

		PathRenderer->RenderColoredPath(CachedGrid, Result, NodeUnitCosts, MinU, MaxU);
	}

	OnPathRecomputed.Broadcast(Result.TotalLength_m, Result.TotalCost, Ms, Result.NodeIndices.Num());
}

// ─────────────────────────────────────────────────────────────────
//  Сетка
// ─────────────────────────────────────────────────────────────────

FRailGridData ARailOptimaGameMode::BuildGridFromTerrain(
	const AProceduralTerrain& InTerrain, float InTargetStep_m) const
{
	const FRailHeightMapData& MapData = InTerrain.GetHeightMapData();
	FRailGridData Grid;
	if (MapData.Width <= 1 || MapData.Height <= 1 || MapData.Heights.Num() == 0) return Grid;

	// Работаем в системе координат ЛАНДШАФТА, а не PNG.
	// Размер карты в метрах = MeshResolution × StepSize_m.
	const int32 TerrainRes = InTerrain.TerrainParameters.MeshResolution;
	const float TerrainStep = InTerrain.TerrainParameters.StepSize_m;
	const float MapSizeMeters = InTerrain.GetMapSizeMeters();

	if (TerrainRes <= 1 || TerrainStep <= 0.0f || MapSizeMeters <= 0.0f)
	{
		UE_LOG(LogTemp, Warning,
			TEXT("[RailGM] Некорректные параметры ландшафта: Resolution=%d, Step=%.2f."),
			TerrainRes, TerrainStep);
		return Grid;
	}

	// Сколько узлов сетки A* укладывается в размер ландшафта.
	// Шаг сетки A* округляем к размеру ландшафта.
	const int32 GridW = FMath::Max(2, FMath::RoundToInt(MapSizeMeters / FMath::Max(InTargetStep_m, 1.0f)));
	const int32 GridH = GridW;
	const float ActualStep_m = MapSizeMeters / static_cast<float>(GridW - 1);

	const float ElevRange = FMath::Max(MapData.MaxElevation_m - MapData.MinElevation_m, 1.0f);

	Grid.Width = GridW;
	Grid.Height = GridH;
	Grid.StepSize_m = ActualStep_m;
	Grid.Nodes.SetNum(GridW * GridH);

	// Соотношение между сеткой A* и PNG (для сэмплирования высот).
	const float ScaleX = static_cast<float>(MapData.Width - 1) / FMath::Max(GridW - 1, 1);
	const float ScaleY = static_cast<float>(MapData.Height - 1) / FMath::Max(GridH - 1, 1);

	for (int32 Y = 0; Y < GridH; ++Y)
	{
		for (int32 X = 0; X < GridW; ++X)
		{
			const int32 SrcX = FMath::Clamp(FMath::RoundToInt(X * ScaleX), 0, MapData.Width - 1);
			const int32 SrcY = FMath::Clamp(FMath::RoundToInt(Y * ScaleY), 0, MapData.Height - 1);

			const float NormH = MapData.Heights[SrcY * MapData.Width + SrcX];
			const float Height_m = MapData.MinElevation_m + NormH * ElevRange;

			FRailGridNode& Node = Grid.Nodes[Grid.IndexOf(X, Y)];
			Node.X = X;
			Node.Y = Y;
			Node.Height_m = Height_m;
			Node.GroundType = ERailGroundType::Loam;
			Node.bIsForbidden = false;
		}
	}

	UE_LOG(LogTemp, Log,
		TEXT("[RailGM] Сетка A*: %dx%d, шаг=%.1f м, размер=%.1f км (совпадает с ландшафтом)."),
		GridW, GridH, ActualStep_m, MapSizeMeters / 1000.0f);

	return Grid;
}

FRailGridData ARailOptimaGameMode::GenerateSyntheticGrid(int32 InSize, float InStepSize_m) const
{
	FRailGridData Grid;
	Grid.Width = InSize;
	Grid.Height = InSize;
	Grid.StepSize_m = InStepSize_m;
	Grid.Nodes.SetNum(InSize * InSize);

	FRandomStream Random(1337);
	for (int32 Y = 0; Y < InSize; ++Y)
	{
		for (int32 X = 0; X < InSize; ++X)
		{
			FRailGridNode& Node = Grid.Nodes[Grid.IndexOf(X, Y)];
			Node.X = X;
			Node.Y = Y;
			const float FX = static_cast<float>(X) / InSize;
			const float FY = static_cast<float>(Y) / InSize;
			Node.Height_m = 200.0f
				+ FMath::Sin(FX * 4.0f * PI) * 80.0f
				+ FMath::Cos(FY * 3.0f * PI) * 60.0f
				+ Random.FRandRange(-5.0f, 5.0f);
		}
	}

	UE_LOG(LogTemp, Warning, TEXT("[RailGM] Синтетическая сетка %dx%d."), InSize, InSize);
	return Grid;
}

// ─────────────────────────────────────────────────────────────────
//  World ↔ Node
// ─────────────────────────────────────────────────────────────────

int32 ARailOptimaGameMode::WorldToNodeIndex(
	const FRailGridData& InGrid, const FVector& InWorldPos) const
{
	if (!PathRenderer || InGrid.Width < 2 || InGrid.Height < 2) return INDEX_NONE;

	const FVector Origin = PathRenderer->GridWorldOrigin;
	const float Step_cm = InGrid.StepSize_m * 100.0f;

	const int32 X = FMath::Clamp(
		FMath::RoundToInt((InWorldPos.X - Origin.X) / Step_cm), 0, InGrid.Width - 1);
	const int32 Y = FMath::Clamp(
		FMath::RoundToInt((InWorldPos.Y - Origin.Y) / Step_cm), 0, InGrid.Height - 1);

	return InGrid.IndexOf(X, Y);
}

FVector ARailOptimaGameMode::NodeIndexToWorldXY(
	const FRailGridData& InGrid, int32 InNodeIndex) const
{
	if (!InGrid.Nodes.IsValidIndex(InNodeIndex) || !PathRenderer) return FVector::ZeroVector;

	const FRailGridNode& Node = InGrid.Nodes[InNodeIndex];
	const FVector Origin = PathRenderer->GridWorldOrigin;
	const float Step_cm = InGrid.StepSize_m * 100.0f;

	return FVector(Origin.X + Node.X * Step_cm, Origin.Y + Node.Y * Step_cm, 0.0f);
}

void ARailOptimaGameMode::HandleEndpointMoved(ARailEndpoint* /*Endpoint*/)
{
	if (RecomputeDebounceSec <= 0.0f) { RecomputePath(); return; }

	GetWorldTimerManager().ClearTimer(RecomputeTimer);
	GetWorldTimerManager().SetTimer(RecomputeTimer, this,
		&ARailOptimaGameMode::HandleDeferredRecompute, RecomputeDebounceSec, false);
}

void ARailOptimaGameMode::HandleDeferredRecompute()
{
	RecomputePath();
}
